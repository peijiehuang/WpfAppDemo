using System.Collections.Generic;
using System.IO;
using System.Linq;
using Microsoft.Extensions.Configuration;
using {Namespace}.Models;
using {Namespace}.Common;
using SqlSugar;
using MiniExcelLibs;
using IConfiguration = Microsoft.Extensions.Configuration.IConfiguration;

namespace {Namespace}.Services
{
    /// <summary>
    /// {TableName} 业务服务实现
    /// </summary>
    public class {TableName}Service : ServiceBase<{TableName}>, I{TableName}Service
    {
        public {TableName}Service(IConfiguration configuration) : base(configuration)
        {
        }

        /// <summary>
        /// 业务逻辑：分页查询
        /// </summary>
        public IEnumerable<{TableName}> Get{TableName}s(int pageIndex, int pageSize, string? keyword, ref int totalCount)
        {
            return Db.Queryable<{TableName}>()
                .WhereIF(!string.IsNullOrEmpty(keyword), it => {SearchLogic})
                .ToPageList(pageIndex, pageSize, ref totalCount);
        }

        /// <summary>
        /// 业务逻辑：数据导出
        /// </summary>
        public MemoryStream Export{TableName}s(string? keyword = null)
        {
            var list = Db.Queryable<{TableName}>()
                .WhereIF(!string.IsNullOrEmpty(keyword), it => {SearchLogic})
                .ToList();
            var stream = new MemoryStream();
            stream.SaveAs(list);
            stream.Seek(0, SeekOrigin.Begin);
            return stream;
        }
    }
}
