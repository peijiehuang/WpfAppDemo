using System;
using System.Collections.ObjectModel;
using System.Threading.Tasks;
using System.IO;
using Microsoft.Win32;
using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;
using Prism.Regions;
using Serilog;
using {Namespace}.Common;
using {Namespace}.Models;
using {Namespace}.Services;

namespace {Namespace}.ViewModels
{
    /// <summary>
    /// {TableName} 列表界面视图模型
    /// </summary>
    public partial class {TableName}ListViewModel : ViewModelBase
    {
        private readonly I{TableName}Service _{camelTableName}Service;
        private readonly IRegionManager _regionManager;
        private readonly IBusyService _busyService;
        private readonly IMessageService _messageService;
        
        [ObservableProperty]
        private string _searchText = string.Empty;

        [ObservableProperty]
        [NotifyCanExecuteChangedFor(nameof(PrevPageCommand))]
        [NotifyCanExecuteChangedFor(nameof(NextPageCommand))]
        private int _pageIndex = 1;

        [ObservableProperty]
        [NotifyCanExecuteChangedFor(nameof(NextPageCommand))]
        private int _pageSize = 20;

        [ObservableProperty]
        [NotifyCanExecuteChangedFor(nameof(NextPageCommand))]
        private int _totalCount;

        /// <summary>
        /// 数据集合
        /// </summary>
        public ObservableCollection<{TableName}> {TableName}s { get; } = new();

        /// <summary>
        /// 构造函数
        /// </summary>
        public {TableName}ListViewModel(I{TableName}Service {camelTableName}Service, IRegionManager regionManager, IBusyService busyService, IMessageService messageService)
        {
            _{camelTableName}Service = {camelTableName}Service;
            _regionManager = regionManager;
            _busyService = busyService;
            _messageService = messageService;
            Title = "{TableName} 管理";
        }

        partial void OnPageIndexChanged(int value)
        {
            // 由命令或显式调用控制加载
        }

        partial void OnPageSizeChanged(int value)
        {
            PageIndex = 1;
            _ = LoadDataAsync();
        }

        /// <summary>
        /// 加载数据
        /// </summary>
        [RelayCommand]
        public async Task LoadDataAsync()
        {
            try
            {
                _busyService.Busy("正在查询...");
                await Task.Delay(50); 
                
                int total = 0;
                var data = await Task.Run(() => _{camelTableName}Service.Get{TableName}s(PageIndex, PageSize, SearchText, ref total));
                
                TotalCount = total;
                {TableName}s.Clear();
                if (data != null)
                {
                    foreach (var item in data)
                    {
                        {TableName}s.Add(item);
                    }
                }
            }
            catch (Exception ex)
            {
                Log.Error(ex, "查询 {TableName} 列表失败");
                await _messageService.ShowMessageAsync($"查询失败: {ex.Message}", "Common_Error");
            }
            finally
            {
                _busyService.Idle();
            }
        }

        /// <summary>
        /// 搜索
        /// </summary>
        [RelayCommand]
        private async Task Search()
        {
            PageIndex = 1;
            await LoadDataAsync();
        }

        /// <summary>
        /// 导出
        /// </summary>
        [RelayCommand]
        private async Task Export()
        {
            try
            {
                var sfd = new SaveFileDialog { Filter = "Excel 文件 (*.xlsx)|*.xlsx", FileName = $"{TableName}导出_{DateTime.Now:yyyyMMddHHmm}" };
                if (sfd.ShowDialog() == true)
                {
                    _busyService.Busy("正在导出...");
                    await Task.Run(() => {
                        using var stream = _{camelTableName}Service.Export{TableName}s(SearchText);
                        using var fileStream = File.Create(sfd.FileName);
                        stream.CopyTo(fileStream);
                    });
                    _busyService.Idle();
                    await _messageService.ShowMessageAsync("数据已成功导出", "Common_Info");
                }
            }
            catch (Exception ex)
            {
                _busyService.Idle();
                await _messageService.ShowMessageAsync($"导出失败: {ex.Message}", "Common_Error");
            }
        }

        [RelayCommand(CanExecute = nameof(CanPrevPage))]
        private async Task PrevPage() 
        {
            PageIndex--;
            await LoadDataAsync();
        }
        private bool CanPrevPage() => PageIndex > 1;

        [RelayCommand(CanExecute = nameof(CanNextPage))]
        private async Task NextPage() 
        {
            PageIndex++;
            await LoadDataAsync();
        }
        private bool CanNextPage() => PageIndex * PageSize < TotalCount;

        [RelayCommand]
        private void Add() => _regionManager.RequestNavigate("ContentRegion", "{TableName}EditView");

        [RelayCommand]
        private void Edit({TableName} entity) => _regionManager.RequestNavigate("ContentRegion", "{TableName}EditView", new NavigationParameters { { "{TableName}", entity } });

        [RelayCommand]
        private async Task Delete({TableName} entity)
        {
            if (await _messageService.ShowConfirmationAsync("确定要删除这条记录吗?", "Common_DeleteConfirm"))
            {
                try
                {
                    _{camelTableName}Service.Delete{TableName}(entity.Id);
                    await LoadDataAsync();
                }
                catch (Exception ex)
                {
                    await _messageService.ShowMessageAsync($"删除失败: {ex.Message}", "Common_Error");
                }
            }
        }

        public override void OnNavigatedTo(NavigationContext navigationContext)
        {
            base.OnNavigatedTo(navigationContext);
            _ = LoadDataAsync();
        }
    }
}
