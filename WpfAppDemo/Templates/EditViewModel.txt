using System.Windows;
using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;
using Prism.Regions;
using {Namespace}.Common;
using {Namespace}.Models;
using {Namespace}.Services;

namespace {Namespace}.ViewModels
{
    /// <summary>
    /// {TableName} 编辑界面视图模型
    /// </summary>
    public partial class {TableName}EditViewModel : ViewModelBase
    {
        private readonly I{TableName}Service _{camelTableName}Service;
        private readonly IRegionManager _regionManager;
        private bool _isEditMode;

        [ObservableProperty]
        [NotifyPropertyChangedFor(nameof(DisplayText))]
        private {TableName} _entity = new {TableName}();

        /// <summary>
        /// 动态标题
        /// </summary>
        public string DisplayText => _isEditMode ? "编辑 {TableName}" : "新增 {TableName}";

        /// <summary>
        /// 构造函数
        /// </summary>
        public {TableName}EditViewModel(I{TableName}Service {camelTableName}Service, IRegionManager regionManager)
        {
            _{camelTableName}Service = {camelTableName}Service;
            _regionManager = regionManager;
        }

        /// <summary>
        /// 保存命令
        /// </summary>
        [RelayCommand]
        private void Save()
        {
            if (Entity == null) return;

            if (_isEditMode)
                _{camelTableName}Service.Update{TableName}(Entity);
            else
                _{camelTableName}Service.Add{TableName}(Entity);

            NavigateBack();
        }

        /// <summary>
        /// 取消命令
        /// </summary>
        [RelayCommand]
        private void Cancel()
        {
            NavigateBack();
        }

        private void NavigateBack()
        {
            _regionManager.RequestNavigate("ContentRegion", "{TableName}ListView");
        }

        public override void OnNavigatedTo(NavigationContext navigationContext)
        {
            base.OnNavigatedTo(navigationContext);

            if (navigationContext.Parameters.ContainsKey("{TableName}"))
            {
                var entity = navigationContext.Parameters.GetValue<{TableName}>("{TableName}");
                if (entity != null)
                {
                    Entity = entity;
                    _isEditMode = true;
                    return;
                }
            }
            
            Entity = new {TableName}();
            _isEditMode = false;
        }
    }
}
